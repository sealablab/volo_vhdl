### Key Implementation Details:

#### Register Mapping Strategy
Since VOLO only supports 8-bit registers, map 16-bit values as follows:
- **CR27-28**: Trigger threshold (split high/low bytes)
- **CR29-30**: Intensity value (split high/low bytes)
- **CR23**: Dual purpose - [7:4] clock divider, [3:0] delay upper nibble
- **CR24**: Delay lower byte (total 12-bit delay counter)

### Starter VHDL Template for volo_main

The generated `DS1120_PD_volo_main.vhd` should include:

```vhdl
library IEEE;
use IEEE.std_logic_1164.all;
use IEEE.numeric_std.all;

-- Volo packages
use work.volo_common_pkg.all;
use work.volo_voltage_pkg.all;

entity DS1120_PD_volo_main is
    port (
        -- Standard volo interface
        Clk       : in  std_logic;
        Reset     : in  std_logic;
        Enable    : in  std_logic;
        ClkEn     : in  std_logic;

        -- App registers (from shim)
        armed         : in  std_logic;
        force_fire    : in  std_logic;
        reset_fsm     : in  std_logic;
        -- ... other register inputs

        -- BRAM interface (required by volo standard)
        bram_addr : out std_logic_vector(11 downto 0);
        bram_data : in  std_logic_vector(31 downto 0);
        bram_we   : out std_logic;

        -- Application outputs
        trigger_out   : out signed(15 downto 0);
        intensity_out : out signed(15 downto 0);
        status_reg    : out std_logic_vector(15 downto 0)
    );
end entity;

architecture rtl of DS1120_PD_volo_main is
    -- FSM states
    constant STATE_READY    : std_logic_vector(2 downto 0) := "000";
    constant STATE_ARMED    : std_logic_vector(2 downto 0) := "001";
    constant STATE_FIRING   : std_logic_vector(2 downto 0) := "010";
    constant STATE_COOLING  : std_logic_vector(2 downto 0) := "011";
    constant STATE_DONE     : std_logic_vector(2 downto 0) := "100";
    constant STATE_TIMEDOUT : std_logic_vector(2 downto 0) := "101";

    -- Voltage constants
    constant MAX_INTENSITY_3V : signed(15 downto 0) := x"4CCD";  -- 3.0V limit

    signal current_state : std_logic_vector(2 downto 0);
    signal divided_clk_en : std_logic;

begin
    -- Clock divider instance
    U_CLK_DIV: entity work.volo_clk_divider
        generic map (MAX_DIV => 16)
        port map (
            clk     => Clk,
            rst_n   => not Reset,
            enable  => Enable,
            div_sel => "0000" & clk_div_bits,  -- From register
            clk_en  => divided_clk_en,
            stat_reg => open
        );

    -- FSM process placeholder
    FSM: process(Clk, Reset)
    begin
        if Reset = '1' then
            current_state <= STATE_READY;
            trigger_out <= (others => '0');
            intensity_out <= (others => '0');
        elsif rising_edge(Clk) then
            if divided_clk_en = '1' and Enable = '1' then
                -- FSM logic here
            end if;
        end if;
    end process;

    -- Status register assembly
    status_reg(15 downto 13) <= current_state;
    status_reg(12 downto 0) <= (others => '0');  -- TODO: Add status bits

end architecture;
```
